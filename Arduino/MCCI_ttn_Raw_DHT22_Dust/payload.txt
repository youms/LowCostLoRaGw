// TTN Payload Decoder — MCCI_ttn_Raw_DHT22_Dust
// Sensor: DHT22 + GP2Y1010AU0F optical dust sensor
//
// Payload layout (6 bytes, positional):
//   [0-1]  Temperature  int16  ×100  → °C
//   [2-3]  Humidity     uint16 ×100  → %
//   [4-5]  Dust ADC     uint16       → raw 0–1023 (averaged over 10 samples)
//
// Dust density formula (Sharp GP2Y1010AU0F datasheet):
//   voltage    = ADC × (5.0 / 1023)
//   dust µg/m³ = max(0,  170 × voltage − 100)
//
// Pin note: LED control is on D5 (D2 is reserved for LMIC DIO0)

function Decoder(bytes, port) {
  var temp = (bytes[0] << 8) | bytes[1];
  if (temp > 32767) temp -= 65536; // int16 sign correction

  var hum     = ((bytes[2] << 8) | bytes[3]) / 100.0;
  var adc     = (bytes[4] << 8) | bytes[5];
  var voltage = adc * (5.0 / 1023.0);
  var dust    = Math.max(0, 170.0 * voltage - 100.0);

  return {
    temperature  : temp / 100.0,
    humidity     : hum,
    dust_adc     : adc,
    dust_voltage : parseFloat(voltage.toFixed(3)),
    dust_ugm3    : parseFloat(dust.toFixed(1))
  };
}
